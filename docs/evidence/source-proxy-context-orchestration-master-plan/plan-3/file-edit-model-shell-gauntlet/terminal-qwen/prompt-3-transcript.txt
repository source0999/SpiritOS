First, make sure you have Flask installed. You can install it using pip if [K

Now, let's create a file named `app.py` with the following content:

```python
from flask import Flask, request, jsonify

app = Flask(__name__)

@app.route('/add', methods=['GET'])
def add():
    num1 = float(request.args.get('num1'))
    num2 = float(request.args.get('num2'))
    result = num1 + num2
    return jsonify({'result': result})

@app.route('/subtract', methods=['GET'])
def subtract():
    num1 = float(request.args.get('num1'))
    num2 = float(request.args.get('num2'))
    result = num1 - num2
    return jsonify({'result': result})

@app.route('/multiply', methods=['GET'])
def multiply():
    num1 = float(request.args.get('num1'))
    num2 = float(request.args.get('num2'))
    result = num1 * num2
    return jsonify({'result': result})

@app.route('/divide', methods=['GET'])
def divide():
    num1 = float(request.args.get('num1'))
    num2 = float(request.args.get('num2'))
    if num2 == 0:
        return jsonify({'error': 'Cannot divide by zero'}), 400
    result = num1 / num2
    return jsonify({'result': result})

if __name__ == '__main__':
    app.run(debug=True)
```

`/multiply`, and `/divide`. Each route takes two query parameters, `num1` a[1D[K
and `num2`, performs the corresponding arithmetic operation, and returns th[2D[K
the result in JSON format.
